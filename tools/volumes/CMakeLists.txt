cmake_minimum_required(VERSION 3.24)
project(kgfx_volume_converter LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "The VDB converter runs on the host; it is never linked into game or PS4 binaries")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "${VOLUME_DEPENDENCY_PREFIX}/lib/cmake/OpenVDB")
list(APPEND CMAKE_PREFIX_PATH "${VOLUME_DEPENDENCY_PREFIX}")
find_package(OpenVDB 12.1.1 EXACT REQUIRED COMPONENTS openvdb)
add_executable(kgfx_volume_converter volume_converter.cpp)
target_link_libraries(kgfx_volume_converter PRIVATE OpenVDB::openvdb)
