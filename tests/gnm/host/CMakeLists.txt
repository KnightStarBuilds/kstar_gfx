cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_C_COMPILER)
    find_program(KGFX_XCRUN xcrun REQUIRED)
    execute_process(COMMAND "${KGFX_XCRUN}" --find clang
        OUTPUT_VARIABLE CMAKE_C_COMPILER OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
endif()
project(kstar_gfx_gnm_layout LANGUAGES C)
option(KSTAR_GFX_OFFLINE "Require cached, verified dependencies" OFF)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja Multi-Config")
    message(FATAL_ERROR "Use Ninja Multi-Config and --config Debug or Release")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/Gnm.cmake")
kgfx_prepare_opengnm_source(opengnm_source)
set(OPENGNM_PLATFORM generic CACHE STRING "" FORCE)
set(OPENGNM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OPENGNM_SUBMIT_WRAPPERS OFF CACHE BOOL "" FORCE)
add_subdirectory("${opengnm_source}" "${CMAKE_CURRENT_BINARY_DIR}/opengnm")
add_executable(kgfx_gnm_layout ../layout.c)
target_link_libraries(kgfx_gnm_layout PRIVATE opengnm::opengnm)
enable_testing()
add_test(NAME kgfx_gnm_layout COMMAND kgfx_gnm_layout)
